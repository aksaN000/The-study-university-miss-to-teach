% ---- Type: one warm, well-built serif family carries the whole book.
% Weight and size build hierarchy, not a mismatched second typeface.
% Lora ships as a variable font; xdvipdfmx mis-embeds it when fontspec
% resolves it through fontconfig's named-instance lookup (a known class
% of variable-font bug), so it's loaded here by explicit file path
% instead, which sidesteps that path entirely and is more portable
% besides -- it doesn't depend on Lora being registered with the OS. ----
\setmainfont{Lora}[
  Path = fonts/,
  UprightFont = Lora-Variable.ttf,
  BoldFont = Lora-Variable.ttf,
  ItalicFont = Lora-Italic-Variable.ttf,
  BoldItalicFont = Lora-Italic-Variable.ttf,
]
\setmonofont{Latin Modern Mono}[Scale=0.92]
% \displayfont is Lora at whatever weight is invoked around it (bfseries
% for headings) -- kept as a named alias so heading commands read clearly
% and so a different display face can be swapped in later in one place.
\let\displayfont\normalfont
